USERS(user_id PK, full_name, email UNIQUE, phone UNIQUE, created_at)
VENUES(venue_id PK, venue_name, city, address)
SCREENS(screen_id PK, venue_id FK->VENUES, screen_name, total_seats, UNIQUE(venue_id,screen_name))
SEATS(seat_id PK, screen_id FK->SCREENS, seat_row, seat_number, seat_type, UNIQUE(screen_id,seat_row,seat_number))
EVENTS(event_id PK, event_name, event_type, duration_minutes, description)
SHOWS(show_id PK, event_id FK->EVENTS, screen_id FK->SCREENS, show_date, start_time, base_price, status, UNIQUE(screen_id,show_date,start_time))
BOOKINGS(booking_id PK, user_id FK->USERS, show_id FK->SHOWS, booking_time, booking_status, total_amount)
BOOKING_SEATS(booking_seat_id PK, booking_id FK->BOOKINGS, seat_id FK->SEATS, price, UNIQUE(booking_id,seat_id))
PAYMENTS(payment_id PK, booking_id FK->BOOKINGS UNIQUE, payment_method, amount, payment_status, paid_at)
TICKETS(ticket_id PK, booking_id FK->BOOKINGS, ticket_code UNIQUE, issued_at)
